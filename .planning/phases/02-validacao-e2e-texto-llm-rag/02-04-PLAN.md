Criar workflow GitHub Actions dedicado para testes RAG E2E.

**Workflow structure:**
```yaml
name: RAG E2E Tests

on:
  pull_request:
    branches: [main, develop]
    paths:
      - 'src/modules/knowledge/**'
      - 'src/modules/llm/**'
      - 'test/rag-*.e2e-spec.ts'
      - 'test/fixtures/rag-*.json'
      - 'database/tests/validate_rag_retrieval.py'
      - '.github/workflows/rag-e2e.yml'
  push:
    branches: [main]
  workflow_dispatch:

jobs:
  rag-e2e:
    name: RAG E2E Tests
    runs-on: ubuntu-latest
    timeout-minutes: 15

    services:
      postgres:
        image: pgvector/pgvector:pg16
        env:
          POSTGRES_USER: openwa
          POSTGRES_PASSWORD: test_password
          POSTGRES_DB: openwa
        ports:
          - 5432:5432
        options: >-
          --health-cmd pg_isready
          --health-interval 10s
          --health-timeout 5s
          --health-retries 5

      redis:
        image: redis:7-alpine
        ports:
          - 6379:6379
        options: >-
          --health-cmd "redis-cli ping"
          --health-interval 10s
          --health-timeout 5s
          --health-retries 5

    steps:
      - name: Checkout code
        uses: actions/checkout@v4

      - name: Setup Node.js 22
        uses: actions/setup-node@v4
        with:
          node-version: '22'
          cache: 'npm'

      - name: Setup Python 3.11
        uses: actions/setup-python@v5
        with:
          python-version: '3.11'
          cache: 'pip'

      - name: Install dependencies
        run: |
          npm ci
          pip install psycopg2-binary numpy

      - name: Wait for PostgreSQL
        run: |
          until pg_isready -h localhost -p 5432 -U openwa; do
            echo "Waiting for PostgreSQL..."
            sleep 2
          done

      - name: Create pgvector extension
        run: |
          PGPASSWORD=test_password psql -h localhost -U openwa -d openwa -c "CREATE EXTENSION IF NOT EXISTS vector;"

      - name: Run database migrations
        run: npm run migration:run
        env:
          DATABASE_TYPE: postgres
          DATABASE_HOST: localhost
          DATABASE_PORT: 5432
          DATABASE_USERNAME: openwa
          DATABASE_PASSWORD: test_password
          DATABASE_NAME: openwa

      - name: Seed test fixtures
        run: |
          PGPASSWORD=test_password psql -h localhost -U openwa -d openwa -f database/tests/fixtures/seed_test_faq.sql

      - name: Run RAG E2E tests
        run: npm run test:e2e:rag
        env:
          NODE_ENV: test
          DATABASE_TYPE: postgres
          DATABASE_HOST: localhost
          DATABASE_PORT: 5432
          DATABASE_USERNAME: openwa
          DATABASE_PASSWORD: test_password
          DATABASE_NAME: openwa
          REDIS_HOST: localhost
          REDIS_PORT: 6379
          GROQ_API_KEY: ${{ secrets.GROQ_API_KEY }}
          OPENAI_API_KEY: ${{ secrets.OPENAI_API_KEY }}

      - name: Run Python RAG metrics
        run: python3 database/tests/validate_rag_retrieval.py
        env:
          PGHOST: localhost
          PGPORT: 5432
          PGUSER: openwa
          PGPASSWORD: test_password
          PGDATABASE: openwa

      - name: Upload test results
        if: always()
        uses: actions/upload-artifact@v4
        with:
          name: rag-test-results
          path: |
            test-results/
            coverage/
          retention-days: 7
```

**Key features:**
- Trigger em PRs que modificam código RAG/LLM/testes
- Service containers: pgvector/pgvector:pg16 + redis:7-alpine
- Setup Node 22 + Python 3.11
- Seed fixtures antes dos testes
- Executa testes E2E TypeScript + métricas Python
- Upload de artifacts para debug
