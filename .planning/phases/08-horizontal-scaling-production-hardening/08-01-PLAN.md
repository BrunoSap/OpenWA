    ```nginx
    upstream openwa_backend {
        ip_hash;  # Sticky sessions via IP
        server openwa-api-1:2785;
        server openwa-api-2:2785;
    }

    server {
        listen 80;
        
        location / {
            proxy_pass http://openwa_backend;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Replica $upstream_addr;  # Debug: qual replica respondeu
        }
        
        # Health checks diretos (sem sticky)
        location /api/health {
            proxy_pass http://openwa_backend;
            proxy_next_upstream error timeout http_503;
        }
    }
    ```

    **Reasoning:** ip_hash mantém afinidade baseada em IP do cliente. X-Replica header permite teste E2E validar sticky behavior.

    **Alternatives rejected:**
    - Cookie-based sticky: requer nginx-plus (comercial) ou módulo third-party
    - Random load balancing: quebra engine state in-process

    **src/modules/health/indicators/redis.health.ts:**
    ```typescript
    import { Injectable } from '@nestjs/common';
    import { HealthIndicator, HealthIndicatorResult, HealthCheckError } from '@nestjs/terminus';
    import Redis from 'ioredis';

    @Injectable()
    export class RedisHealthIndicator extends HealthIndicator {
      constructor(private readonly redisClient: Redis) {
        super();
      }

      async isHealthy(key: string): Promise<HealthIndicatorResult> {
        try {
          const pong = await this.redisClient.ping();
          if (pong !== 'PONG') {
            throw new HealthCheckError('Redis PING unexpected response', { redis: { status: 'down', response: pong } });
          }
          return this.getStatus(key, true, { status: 'up' });
        } catch (error) {
          throw new HealthCheckError('Redis unreachable', { redis: { status: 'down', error: error.message } });
        }
      }
    }
    ```

    **src/modules/health/indicators/engine.health.ts:**
    ```typescript
    import { Injectable } from '@nestjs/common';
    import { HealthIndicator, HealthIndicatorResult, HealthCheckError } from '@nestjs/terminus';
    import { SessionService } from '../../session/session.service';

    @Injectable()
    export class EngineHealthIndicator extends HealthIndicator {
      constructor(private readonly sessionService: SessionService) {
        super();
      }

      async isHealthy(key: string): Promise<HealthIndicatorResult> {
        try {
          // Check if SessionService não está travado — timeout 3s
          const timeoutPromise = new Promise((_, reject) => 
            setTimeout(() => reject(new Error('SessionService timeout')), 3000)
          );
          const sessionsPromise = this.sessionService.getAllSessions();
          
          await Promise.race([sessionsPromise, timeoutPromise]);
          
          return this.getStatus(key, true, { status: 'up' });
        } catch (error) {
          throw new HealthCheckError('Engine not ready', { engine: { status: 'down', error: error.message } });
        }
      }
    }
    ```

    **test/multi-replica.e2e-spec.ts:**
    ```typescript
    import { Test, TestingModule } from '@nestjs/testing';
    import { INestApplication } from '@nestjs/common';
    import * as request from 'supertest';
    import { AppModule } from '../src/app.module';

    describe('Multi-Replica Scaling (e2e)', () => {
      let app: INestApplication;

      beforeAll(async () => {
        const moduleFixture: TestingModule = await Test.createTestingModule({
          imports: [AppModule],
        }).compile();

        app = moduleFixture.createNestApplication();
        await app.init();
      });

      afterAll(async () => {
        await app.close();
      });

      it('Sticky sessions: 50 sequential requests from same IP go to same replica', async () => {
        const responses = [];
        
        // Loop 50 requests
        for (let i = 0; i < 50; i++) {
          const res = await request('http://localhost:2785')
            .get('/api/health/live')
            .set('X-Forwarded-For', '192.168.1.100');  // Simulate same client IP
          
          responses.push(res.headers['x-replica']);
        }

        // All responses should have same X-Replica value (sticky)
        const uniqueReplicas = [...new Set(responses)];
        expect(uniqueReplicas.length).toBe(1);
      });

      it('Load distribution: requests from different IPs use different replicas', async () => {
        const replicas = new Set();

        // 10 requests, cada um de IP diferente
        for (let i = 0; i < 10; i++) {
          const res = await request('http://localhost:2785')
            .get('/api/health/live')
            .set('X-Forwarded-For', `192.168.1.${100 + i}`);
          
          replicas.add(res.headers['x-replica']);
        }

        // Deve ter usado ambas replicas (ip_hash distribui)
        expect(replicas.size).toBeGreaterThanOrEqual(2);
      });

      it('Health check removes unhealthy replica from pool', async () => {
        // Simular replica 1 unhealthy (Redis down)
        // TODO: mock Redis failure no container openwa-api-1
        
        // Requests devem ir apenas para replica 2
        const responses = [];
        for (let i = 0; i < 10; i++) {
          const res = await request('http://localhost:2785')
            .get('/api/health/live')
            .set('X-Forwarded-For', '192.168.1.200');
          
          responses.push(res.headers['x-replica']);
        }

        // Todas devem ser replica 2 (openwa-api-2:2785)
        const uniqueReplicas = [...new Set(responses)];
        expect(uniqueReplicas.length).toBe(1);
        expect(uniqueReplicas[0]).toContain('openwa-api-2');
      });
    });
    ```

    **Reasoning:** X-Replica header (set pelo nginx) permite teste validar sticky behavior sem inspecionar nginx logs. Set múltiplo de X-Forwarded-For simula clientes diferentes (nginx ip_hash usa esse header).

    **Test data:** IPs 192.168.1.x simulam clientes reais; nginx ip_hash distribui baseado em hash(IP).

    **Alternatives rejected:**
    - Sticky via cookies: não funciona com ip_hash (nginx OSS), requer nginx-plus
    - Inspecionar nginx logs: frágil, teste deve ser self-contained

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Client → nginx | Untrusted requests, IP spoofing via X-Forwarded-For |
| nginx → replicas | Internal network, sticky sessions enforced |
| Replica → Redis | Shared state, connection pool exhaustion risk |

## STRIDE Threat Register

| Threat ID | Category | Component | Severity | Disposition | Mitigation Plan |
|-----------|----------|-----------|----------|-------------|-----------------|
| T-08-01 | Spoofing | nginx X-Forwarded-For | medium | mitigate | Validate X-Forwarded-For apenas de proxy trusted; set real_ip_header + set_real_ip_from interno |
| T-08-02 | Tampering | Sticky session bypass | low | accept | ip_hash pode falhar se IP muda (mobile); session re-auth é recovery natural |
| T-08-03 | Denial of Service | Redis connection pool | high | mitigate | Connection pooling BullMQ/RedisIoAdapter já implementado; monitor CLIENT LIST |
| T-08-04 | Information Disclosure | X-Replica header | low | accept | Expõe topology mas não secrets; útil para debug |
| T-08-SC | Tampering | npm/pip installs | high | mitigate | Package legitimacy gate já passou (RESEARCH.md §Package Legitimacy Audit) |
